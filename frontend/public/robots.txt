# WeAre Creators
#
# Two kinds of page are public and meant to be found: a brand's own page at
# /brands/{id} and a live brief at /c/{id}. Both are server-rendered by the
# backend precisely so a crawler can read them — see PREVIEW.md for the
# rewrites that make those paths reach it.
#
# Everything else needs an account. Those routes are a static bundle that
# renders nothing without a session, so a crawler would index an empty shell.

User-agent: *

# literally, so this covers /brand/dashboard and leaves /brands/{id} — the
# public page, and the whole point of this file — allowed.

Disallow: /api/
Disallow: /dashboard
Disallow: /onboarding
Disallow: /profile
